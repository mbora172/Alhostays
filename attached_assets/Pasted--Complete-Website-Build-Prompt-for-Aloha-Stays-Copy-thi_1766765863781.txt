## PART 2: AIRBNB CALENDAR SYNC (CRITICAL)

### Requirement:
Prevent double bookings by syncing with Airbnb calendar in real-time.

### Implementation Options:

**Option A: iCal Sync (Recommended - Simpler)**
```javascript
// Airbnb provides iCal URLs for each listing
// Property 1 iCal: https://www.airbnb.com/calendar/ical/LISTING_ID.ics
// Property 2 iCal: https://www.airbnb.com/calendar/ical/LISTING_ID.ics

// Import iCal library
import ical from 'ical';
import axios from 'axios';

// Function to fetch and parse Airbnb calendar
async function getAirbnbBlockedDates(icalUrl) {
  const response = await axios.get(icalUrl);
  const events = ical.parseICS(response.data);
  
  const blockedDates = [];
  for (let event in events) {
    if (events.hasOwnProperty(event)) {
      const ev = events[event];
      if (ev.type === 'VEVENT') {
        // Add check-in to check-out dates as blocked
        blockedDates.push({
          start: ev.start,
          end: ev.end,
          summary: ev.summary
        });
      }
    }
  }
  return blockedDates;
}

// Sync calendars every hour
setInterval(async () => {
  const property1Blocked = await getAirbnbBlockedDates(PROPERTY1_ICAL_URL);
  const property2Blocked = await getAirbnbBlockedDates(PROPERTY2_ICAL_URL);
  
  // Update your database with blocked dates
  await updateBlockedDates('property1', property1Blocked);
  await updateBlockedDates('property2', property2Blocked);
}, 3600000); // Every hour
```

**How to Get iCal URLs:**
1. Go to your Airbnb listing
2. Click "Availability Settings"
3. Scroll to "Calendar sync"
4. Copy the "Export calendar" URL
5. Add to your website config

### Calendar Component Implementation:

```javascript
// Use a calendar library like react-calendar or react-dates
import { Calendar } from 'react-calendar';
import { useState, useEffect } from 'react';

function PropertyCalendar({ propertyId }) {
  const [blockedDates, setBlockedDates] = useState([]);
  
  useEffect(() => {
    // Fetch blocked dates from your database (synced from Airbnb)
    fetch(`/api/blocked-dates/${propertyId}`)
      .then(res => res.json())
      .then(data => setBlockedDates(data));
  }, [propertyId]);
  
  const isDateBlocked = (date) => {
    return blockedDates.some(blocked => 
      date >= new Date(blocked.start) && 
      date <= new Date(blocked.end)
    );
  };
  
  return (
    <Calendar
      tileDisabled={({ date }) => isDateBlocked(date)}
      minDate={new Date()} // Can't book past dates
    />
  );
}
```

```javascript
// Backend: Create payment intent
import Stripe from 'stripe';
const stripe = new Stripe(process.env.STRIPE_SECRET_KEY);

app.post('/api/create-payment-intent', async (req, res) => {
  const { amount, bookingDetails } = req.body;
  
  try {
    const paymentIntent = await stripe.paymentIntents.create({
      amount: amount * 100, // Convert to cents
      currency: 'usd',
      metadata: {
        property: bookingDetails.property,
        checkIn: bookingDetails.checkIn,
        checkOut: bookingDetails.checkOut,
        guests: bookingDetails.guests
      }
    });
    
    res.json({ clientSecret: paymentIntent.client_secret });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});
```

### Payment Form Component:
```javascript
import { CardElement, useStripe, useElements } from '@stripe/react-stripe-js';

function PaymentForm({ bookingDetails, totalAmount }) {
  const stripe = useStripe();
  const elements = useElements();
  const [error, setError] = useState(null);
  const [processing, setProcessing] = useState(false);
  
  const handleSubmit = async (event) => {
    event.preventDefault();
    setProcessing(true);
    
    try {
      // Create payment intent
      const response = await fetch('/api/create-payment-intent', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ amount: totalAmount, bookingDetails })
      });
      
      const { clientSecret } = await response.json();
      
      // Confirm payment
      const result = await stripe.confirmCardPayment(clientSecret, {
        payment_method: {
          card: elements.getElement(CardElement),
          billing_details: {
            name: bookingDetails.guestName,
            email: bookingDetails.guestEmail
          }
        }
      });
      
      if (result.error) {
        setError(result.error.message);
        setProcessing(false);
      } else {
        // Payment successful!
        await createBooking(bookingDetails, result.paymentIntent.id);
        window.location.href = '/booking-confirmation?id=' + result.paymentIntent.id;
      }
    } catch (err) {
      setError(err.message);
      setProcessing(false);
    }
  };
  
  return (
    <form onSubmit={handleSubmit}>
      <CardElement />
      {error && <div className="error">{error}</div>}
      <button disabled={!stripe || processing}>
        {processing ? 'Processing...' : `Pay $${totalAmount}`}
      </button>
    </form>
  );
}
```

```javascript
// When user selects "Pay Later"
async function handlePayLater(bookingDetails) {
  try {
    // Create pending booking in database
    const booking = await fetch('/api/bookings', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        ...bookingDetails,
        status: 'pending_payment',
        paymentMethod: 'pay_later'
      })
    });
    
    const result = await booking.json();
    
    // Send email to owner with booking details
    await fetch('/api/notify-owner', {
      method: 'POST',
      body: JSON.stringify({
        bookingId: result.id,
        property: bookingDetails.property,
        guest: bookingDetails.guestName,
        checkIn: bookingDetails.checkIn,
        checkOut: bookingDetails.checkOut,
        total: bookingDetails.total
      })
    });
    
    // Send confirmation email to guest
    await fetch('/api/send-guest-email', {
      method: 'POST',
      body: JSON.stringify({
        to: bookingDetails.guestEmail,
        subject: 'Booking Request Received - Aloha Stays',
        message: `Thank you for your booking request!
        
        Property: ${bookingDetails.property}
        Check-in: ${bookingDetails.checkIn}
        Check-out: ${bookingDetails.checkOut}
        Total: $${bookingDetails.total}
        
        We'll send you a payment link within 24 hours.
        
        Mahalo!
        Aloha Stays Team`
      })
    });
    
    // Redirect to confirmation page
    window.location.href = '/booking-request-submitted';
    
  } catch (error) {
    console.error('Pay Later error:', error);
    alert('There was an error submitting your request. Please try again.');
  }
}
```

## PART 6: AVAILABILITY CHECK LOGIC

### Before Allowing Booking:
```javascript
async function checkAvailability(propertyId, checkIn, checkOut) {
  // Convert dates
  const checkInDate = new Date(checkIn);
  const checkOutDate = new Date(checkOut);
  
  // Check existing bookings
  const existingBookings = await db.query(`
    SELECT * FROM bookings 
    WHERE property_id = $1 
    AND status IN ('confirmed', 'pending')
    AND (
      (check_in_date <= $2 AND check_out_date > $2) OR
      (check_in_date < $3 AND check_out_date >= $3) OR
      (check_in_date >= $2 AND check_out_date <= $3)
    )
  `, [propertyId, checkInDate, checkOutDate]);
  
  if (existingBookings.length > 0) {
    return { available: false, reason: 'dates_already_booked' };
  }
  
  // Check Airbnb blocked dates
  const blockedDates = await db.query(`
    SELECT * FROM blocked_dates 
    WHERE property_id = $1 
    AND (
      (start_date <= $2 AND end_date > $2) OR
      (start_date < $3 AND end_date >= $3) OR
      (start_date >= $2 AND end_date <= $3)
    )
  `, [propertyId, checkInDate, checkOutDate]);
  
  if (blockedDates.length > 0) {
    return { available: false, reason: 'dates_blocked_airbnb' };
  }
  
  // Check minimum/maximum stay requirements
  const numNights = Math.ceil((checkOutDate - checkInDate) / (1000 * 60 * 60 * 24));
  if (numNights < 2) {
    return { available: false, reason: 'minimum_stay_2_nights' };
  }
  
  return { available: true };
}
```
